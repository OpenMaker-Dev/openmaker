# OnWindowTransaction (file trigger 209) on FileMaker Server's script
# engine: which operations fire it, its JSON parameter, and when it runs
# (findings: results/s35-findings.md, part C).
# Base file: tasks-en, generated, with the s35-portals "Attachment List"
# layout, OnWindowTransaction running "otx" (no field name option, no field
# named OnWindowTransaction), and Tasks::Priority validated Always by
# calculation (Length ( Self ) <= 3) for a Commit Transaction that fails.
# "otx" appends {its parameter} to $$tx and counts the firings in $$txn.
# Each probe clears both after its own setup and returns: the step's error
# (LastError right after the commit or delete), $$tx read by the step
# after it ("seen"), and the run line's $$tx and $$txn after the script.
@layout	Attachment List	Attachments	Assignments
@filetrigger	OnWindowTransaction	otx
@validate	Tasks::Priority	Always	Length ( Self ) <= 3

== otx
Set Variable [ $$tx ; $$tx & "{" & Get ( ScriptParameter ) & "}" ]
Set Variable [ $$txn ; $$txn + 1 ]

== mk
# Parameter: name ¶ n. A new task with n attachments "name-1" ..., then
# Task Details on it (all records); $$tx, $$txn cleared afterwards.
Set Variable [ $name ; GetValue ( Get ( ScriptParameter ) ; 1 ) ]
Set Variable [ $n ; GetAsNumber ( GetValue ( Get ( ScriptParameter ) ; 2 ) ) ]
Go to Layout [ "Task Details" ]
New Record/Request
Set Field [ Tasks::Task ; $name ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $k ; Tasks::PrimaryKey ]
Go to Layout [ "Attachment List" ]
Set Variable [ $i ; 0 ]
Loop
Exit Loop If [ $i >= $n ]
Set Variable [ $i ; $i + 1 ]
New Record/Request
Set Field [ Attachments::ForeignKey ; $k ]
Set Field [ Attachments::Description ; $name & "-" & $i ]
Commit Records/Requests [ With dialog: Off ]
End Loop
Go to Layout [ "Task Details" ]
Show All Records
Go to Record/Request/Page [ First ]
Loop
Exit Loop If [ Tasks::Task = $name ]
Go to Record/Request/Page [ Next ; Exit after last: On ]
End Loop
Set Variable [ $$tx ; "" ]
Set Variable [ $$txn ; 0 ]

== new
# Commit Records of a new record.
Set Error Capture [ On ]
Go to Layout [ "Task Details" ]
Set Variable [ $$tx ; "" ]
New Record/Request
Set Field [ Tasks::Task ; Get ( ScriptParameter ) ]
Set Variable [ $before ; $$tx ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Set Variable [ $seen ; $$tx ]
Exit Script [ "before " & $before & " commit " & $e & " seen " & $seen & " id " & Get ( RecordID ) ]

== edit
# Commit Records of an edit (the task named by the parameter gets " e").
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Set Field [ Tasks::Task ; Tasks::Task & " e" ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Set Variable [ $seen ; $$tx ]
Exit Script [ "commit " & $e & " seen " & $seen & " id " & Get ( RecordID ) ]

== nochange
# Commit Records with nothing changed, then Set Field of the same value.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Set Variable [ $seen ; $$tx ]
Set Field [ Tasks::Task ; Tasks::Task ]
Commit Records/Requests [ With dialog: Off ]
Exit Script [ "commit " & $e & " seen " & $seen & " same value " & Get ( LastError ) & " seen " & $$tx ]

== delete
# Delete Record.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Set Variable [ $id ; Get ( RecordID ) ]
Delete Record/Request [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Set Variable [ $seen ; $$tx ]
Exit Script [ "delete " & $e & " seen " & $seen & " id " & $id ]

== delete all
# Delete All Records on Attachment List after a find for the task's two
# attachments.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶2" ]
Go to Layout [ "Attachment List" ]
Enter Find Mode [ Pause: Off ]
Set Field [ Attachments::Description ; Get ( ScriptParameter ) & "-*" ]
Perform Find
Set Variable [ $f ; Get ( LastError ) & "/" & Get ( FoundCount ) ]
Set Variable [ $$tx ; "" ]
Delete All Records [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Set Variable [ $seen ; $$tx ]
Exit Script [ "find " & $f & " delete all " & $e & " seen " & $seen ]

== tx two
# Commit Transaction of a new task, an edit of an existing task and a new
# attachment.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Set Variable [ $k ; Tasks::PrimaryKey ]
Open Transaction
Set Field [ Tasks::Task ; Tasks::Task & " e" ]
New Record/Request
Set Field [ Tasks::Task ; Get ( ScriptParameter ) & " new" ]
Go to Layout [ "Attachment List" ]
New Record/Request
Set Field [ Attachments::ForeignKey ; $k ]
Set Field [ Attachments::Description ; Get ( ScriptParameter ) & "-a" ]
Set Variable [ $before ; $$tx ]
Commit Transaction
Set Variable [ $e ; Get ( LastError ) ]
Set Variable [ $seen ; $$tx ]
Exit Script [ "before " & $before & " commit tx " & $e & " seen " & $seen ]

== tx delete
# Commit Transaction of a delete.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Open Transaction
Delete Record/Request [ With dialog: Off ]
Set Variable [ $before ; $$tx ]
Commit Transaction
Set Variable [ $e ; Get ( LastError ) ]
Exit Script [ "before " & $before & " commit tx " & $e & " seen " & $$tx ]

== portal delete
# Delete Portal Row of row 1, then Commit Records.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶2" ]
Go to Portal Row [ First ]
Delete Portal Row [ With dialog: Off ]
Set Variable [ $d ; Get ( LastError ) & " seen " & $$tx ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Exit Script [ "delete row " & $d & " commit " & $e & " seen " & $$tx ]

== portal create
# Set Field of a related field without related records (creates one),
# and of the parent, then Commit Records.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Set Field [ Attachments::Description ; Get ( ScriptParameter ) & "-r" ]
Set Field [ Tasks::Description ; "parent" ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Exit Script [ "commit " & $e & " seen " & $$tx ]

== revert record
# A new record reverted, an edit reverted.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Set Field [ Tasks::Task ; "changed" ]
Revert Record/Request [ With dialog: Off ]
New Record/Request
Set Field [ Tasks::Task ; "reverted" ]
Revert Record/Request [ With dialog: Off ]
Exit Script [ "revert " & Get ( LastError ) & " seen " & $$tx ]

== revert tx
# Revert Transaction of a new record.
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Open Transaction
New Record/Request
Set Field [ Tasks::Task ; "tx reverted" ]
Revert Transaction
Commit Transaction
Set Variable [ $e ; Get ( LastError ) ]
Exit Script [ "commit tx " & $e & " seen " & $$tx ]

== failed tx
# A transaction whose commit fails validation (Priority "toolong").
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Open Transaction
New Record/Request
Set Field [ Tasks::Task ; "tx failed" ]
Set Field [ Tasks::Priority ; "toolong" ]
Set Variable [ $s ; Get ( LastError ) ]
Commit Transaction
Set Variable [ $e ; Get ( LastError ) ]
Exit Script [ "set " & $s & " commit tx " & $e & " seen " & $$tx ]

== failed commit
# Commit Records failing validation (no transaction).
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Set Field [ Tasks::Priority ; "toolong" ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Revert Record/Request [ With dialog: Off ]
Exit Script [ "commit " & $e & " seen " & $$tx ]

== two commits
# Two records committed one after the other, then a sub-script commit.
Set Error Capture [ On ]
Go to Layout [ "Task Details" ]
Set Variable [ $$tx ; "" ]
New Record/Request
Set Field [ Tasks::Task ; Get ( ScriptParameter ) & "1" ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $s1 ; $$txn ]
New Record/Request
Set Field [ Tasks::Task ; Get ( ScriptParameter ) & "2" ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $s2 ; $$txn ]
Exit Script [ "after first " & $s1 & " after second " & $s2 & " seen " & $$tx ]

revert tx	rt	$$tx	$$txn

failed commit	fc	$$tx	$$txn
two commits	tc	$$tx	$$txn
