# s35-wintx with a text field named OnWindowTransaction in Tasks (a copy
# of Tasks::Description) and none in Attachments, no field name option:
# the parameter's field contents for text, JSON text and empty, and for a
# table without the field. Scripts and output as s35-wintx.
@layout	Attachment List	Attachments	Assignments
@field	Tasks	OnWindowTransaction	Description
@filetrigger	OnWindowTransaction	otx

== otx
Set Variable [ $$tx ; $$tx & "{" & Get ( ScriptParameter ) & "}" ]
Set Variable [ $$txn ; $$txn + 1 ]

== mk
# Parameter: name ¶ n. A new task with n attachments "name-1" ..., then
# Task Details on it (all records); $$tx, $$txn cleared afterwards.
Set Variable [ $name ; GetValue ( Get ( ScriptParameter ) ; 1 ) ]
Set Variable [ $n ; GetAsNumber ( GetValue ( Get ( ScriptParameter ) ; 2 ) ) ]
Go to Layout [ "Task Details" ]
New Record/Request
Set Field [ Tasks::Task ; $name ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $k ; Tasks::PrimaryKey ]
Go to Layout [ "Attachment List" ]
Set Variable [ $i ; 0 ]
Loop
Exit Loop If [ $i >= $n ]
Set Variable [ $i ; $i + 1 ]
New Record/Request
Set Field [ Attachments::ForeignKey ; $k ]
Set Field [ Attachments::Description ; $name & "-" & $i ]
Commit Records/Requests [ With dialog: Off ]
End Loop
Go to Layout [ "Task Details" ]
Show All Records
Go to Record/Request/Page [ First ]
Loop
Exit Loop If [ Tasks::Task = $name ]
Go to Record/Request/Page [ Next ; Exit after last: On ]
End Loop
Set Variable [ $$tx ; "" ]
Set Variable [ $$txn ; 0 ]

== new value
# Parameter: the value of Tasks::OnWindowTransaction of a new task.
Set Error Capture [ On ]
Go to Layout [ "Task Details" ]
Set Variable [ $$tx ; "" ]
New Record/Request
Set Field [ Tasks::Task ; "v" ]
Set Field [ Tasks::OnWindowTransaction ; Get ( ScriptParameter ) ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $e ; Get ( LastError ) ]
Exit Script [ "commit " & $e & " seen " & $$tx & " id " & Get ( RecordID ) ]

== tx two
# Commit Transaction of a task edit (Tasks::OnWindowTransaction "task value") and a new
# attachment (Description {"b":2}).
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Set Variable [ $k ; Tasks::PrimaryKey ]
Open Transaction
Set Field [ Tasks::OnWindowTransaction ; "task value" ]
Go to Layout [ "Attachment List" ]
New Record/Request
Set Field [ Attachments::ForeignKey ; $k ]
Set Field [ Attachments::Description ; "{\"b\":2}" ]
Commit Transaction
Set Variable [ $e ; Get ( LastError ) ]
Exit Script [ "commit tx " & $e & " seen " & $$tx ]

== delete value
# Delete Record of a task whose Tasks::OnWindowTransaction is "deleted value".
Set Error Capture [ On ]
Perform Script [ "mk" ; Parameter: Get ( ScriptParameter ) & "¶0" ]
Set Field [ Tasks::OnWindowTransaction ; "deleted value" ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $$tx ; "" ]
Delete Record/Request [ With dialog: Off ]
Exit Script [ "delete " & Get ( LastError ) & " seen " & $$tx ]

--- run
new value	hello	$$tx
new value	{"a":1,"c":[true,null]}	$$tx
new value	[1,2	$$tx
new value		$$tx
new value	42	$$tx
tx two	tt	$$tx
delete value	dv	$$tx
