# The "current script" options when another script is started while one is
# paused or running (script-runtime.md 3.7, U5, U6): "@call" makes the
# harness call ScriptRuntimeBase::CallScript(window, B, no parameter,
# newState, queueAtBottom) right after step 2 of A, the way a button,
# trigger or PSOS callback starts a script. newState 0 continue, 1 halt,
# 2 exit, 3 resume, 4 pause, 5 suspend and resume. Base file: contacts-en.
== q paused
Set Variable [ $$log ; "A1" ]
Pause/Resume Script
Set Variable [ $$log ; $$log & " A3" ]
Exit Script [ $$log & " result(" & Get ( ScriptResult ) & ")" ]
== q running
Set Variable [ $$log ; "A1" ]
Set Variable [ $$log ; $$log & " A2" ]
Set Variable [ $$log ; $$log & " A3" ]
Exit Script [ $$log & " result(" & Get ( ScriptResult ) & ")" ]
== q nested
Set Variable [ $$log ; "A1" ]
Perform Script [ "q nested sub" ]
Set Variable [ $$log ; $$log & " A3" ]
Exit Script [ $$log & " result(" & Get ( ScriptResult ) & ")" ]
== q nested sub
Set Variable [ $$log ; $$log & " S1" ]
Set Variable [ $$log ; $$log & " S2" ]
Set Variable [ $$log ; $$log & " S3" ]
== q other
Set Variable [ $$log ; $$log & " B(" & Get ( ScriptParameter ) & ")" ]
Exit Script [ "b" ]
